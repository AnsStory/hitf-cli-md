Translation Providers Implementation Plan ​

For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

Goal: Add Google Translate (Cloud Translation API) and GPT (OpenAI) translation support to hias-cli, rewrite existing Baidu/Tencent providers under a unified abstraction layer.

Architecture: TranslatorProvider base class + concrete providers (baidu, tencent, google, openai) + ProviderManager singleton registry. Fault-tolerant translation via servicePriority fallback. New config structure under translationSetting.

Tech Stack: Node.js, axios (existing), Joi (existing config validation), node:test (testing).

Global Constraints ​

  • Configuration via .hias/setting.json config files only (not env vars or interactive)
  • No backward compatibility with old provider/appId/secretKey fields
  • Multi-provider with priority fallback: servicePriority: ["tencent", "baidu", "openai", "google"]
  • Google: Google Cloud Translation API v2 (REST, https://translation.googleapis.com/language/translate/v2)
  • GPT: OpenAI Chat Completions API with custom endpoint support, batch via prompt, configurable model
  • Use existing axios for HTTP (no new dependencies unless absolutely necessary)
  • All new code follows existing style: CommonJS modules, no comments unless asked, node:test + assert

// lib/translation/providers/provider-manager.js
const { ConfigurationError } = require('./errors');

class ProviderManager {
  constructor() {
    this.providers = new Map();
  }

  register(name, provider) {
    this.providers.set(name, provider);
  }

  get(name) {
    return this.providers.get(name);
  }

  has(name) {
    return this.providers.has(name);
  }

  getRegisteredProviders() {
    return Array.from(this.providers.keys());
  }

  resolveProvider(name, config) {
    const provider = this.providers.get(name);
    if (!provider) {
      throw new ConfigurationError(`Unknown translation provider: ${name}`, name);
    }
    if (provider.configure && config) {
      provider.configure(config);
    }
    return provider;
  }
}

module.exports = ProviderManager;

// lib/translation/providers/baidu-provider.js
const TranslatorProvider = require('./translator-provider');
const { baiduTranslate } = require('../../utils/baidu-translate');
const { ConfigurationError } = require('./errors');

class BaiduProvider extends TranslatorProvider {
  constructor() {
    super();
    this.appId = null;
    this.secretKey = null;
  }

  configure(config) {
    this.appId = config.appId;
    this.secretKey = config.secretKey;
  }

  getName() {
    return 'baidu';
  }

  getSupportedLanguages() {
    return ['zh', 'en', 'ja', 'ko', 'fr', 'es', 'ar', 'de', 'ru', 'pt', 'it', 'nl', 'pl', 'th', 'vi', 'id', 'ms'];
  }

  async translate(texts, from, to, options = {}) {
    if (!this.appId || !this.secretKey) {
      throw new ConfigurationError('Baidu provider requires appId and secretKey', 'baidu', ['appId', 'secretKey']);
    }
    return await baiduTranslate(texts.join('\n'), this.appId, this.secretKey, from, to);
  }
}

module.exports = BaiduProvider;

// lib/translation/providers/google-provider.js
const TranslatorProvider = require('./translator-provider');
const { googleRequest } = require('../../utils/request');
const { ConfigurationError, TranslationError } = require('./errors');

class GoogleProvider extends TranslatorProvider {
  constructor() {
    super();
    this.apiKey = null;
  }

  configure(config) {
    this.apiKey = config.apiKey;
  }

  getName() {
    return 'google';
  }

  getSupportedLanguages() {
    return [
      'zh', 'en', 'ja', 'ko', 'fr', 'es', 'ar', 'de', 'ru', 'pt',
      'it', 'nl', 'pl', 'th', 'vi', 'id', 'ms', 'hi', 'tr', 'uk',
      'cs', 'da', 'el', 'fi', 'hu', 'no', 'ro', 'sv', 'bg', 'ca',
    ];
  }

  async translate(texts, from, to, options = {}) {
    if (!this.apiKey) {
      throw new ConfigurationError('Google provider requires apiKey', 'google', ['apiKey']);
    }

    try {
      const results = [];
      for (const text of texts) {
        const res = await googleRequest.post({
          url: `/language/translate/v2?key=${this.apiKey}`,
          data: {
            q: text,
            source: from.split('-')[0],
            target: to.split('-')[0],
            format: 'text',
          },
        });
        if (res.data && res.data.translations && res.data.translations.length > 0) {
          results.push(res.data.translations[0].translatedText);
        } else {
          results.push(text);
        }
      }
      return results;
    } catch (error) {
      const msg = error.response?.data?.error?.message || error.message;
      throw new TranslationError(`Google API error: ${msg}`, 'google', error);
    }
  }
}

module.exports = GoogleProvider;

// lib/translation/providers/openai-provider.js
const TranslatorProvider = require('./translator-provider');
const { openaiRequest } = require('../../utils/request');
const { ConfigurationError, TranslationError } = require('./errors');

class OpenAIProvider extends TranslatorProvider {
  constructor() {
    super();
    this.apiKey = null;
    this.baseUrl = 'https://api.openai.com';
    this.model = 'gpt-3.5-turbo';
  }

  configure(config) {
    this.apiKey = config.apiKey;
    if (config.baseUrl) this.baseUrl = config.baseUrl;
    if (config.model) this.model = config.model;
  }

  getName() {
    return 'openai';
  }

  getSupportedLanguages() {
    return [
      'zh', 'en', 'ja', 'ko', 'fr', 'es', 'ar', 'de', 'ru', 'pt',
      'it', 'nl', 'pl', 'th', 'vi', 'id', 'ms', 'hi', 'tr', 'uk',
      'cs', 'da', 'el', 'fi', 'hu', 'no', 'ro', 'sv',
    ];
  }

  async translate(texts, from, to, options = {}) {
    if (!this.apiKey) {
      throw new ConfigurationError('OpenAI provider requires apiKey', 'openai', ['apiKey']);
    }

    const LANG_NAMES = {
      zh: 'Chinese', en: 'English', ja: 'Japanese', ko: 'Korean',
      fr: 'French', es: 'Spanish', ar: 'Arabic', de: 'German',
      ru: 'Russian', pt: 'Portuguese', it: 'Italian', nl: 'Dutch',
      pl: 'Polish', th: 'Thai', vi: 'Vietnamese', id: 'Indonesian',
      ms: 'Malay', hi: 'Hindi', tr: 'Turkish', uk: 'Ukrainian',
      cs: 'Czech', da: 'Danish', el: 'Greek', fi: 'Finnish',
      hu: 'Hungarian', no: 'Norwegian', ro: 'Romanian', sv: 'Swedish',
    };

    const fromLang = LANG_NAMES[from.split('-')[0]] || from;
    const toLang = LANG_NAMES[to.split('-')[0]] || to;

    const prompt = `Translate the following texts from ${fromLang} to ${toLang}. Return a JSON array of translated strings, one per input text, maintaining the same order. Do not include any explanation, only the JSON array.\n\nTexts:\n${texts.map((t, i) => `${i + 1}. ${t}`).join('\n')}`;

    try {
      const res = await openaiRequest.post({
        url: `${this.baseUrl.replace(/\/+$/, '')}/v1/chat/completions`,
        headers: {
          Authorization: `Bearer ${this.apiKey}`,
          'Content-Type': 'application/json',
        },
        data: {
          model: this.model,
          messages: [
            { role: 'system', content: 'You are a professional translator. Return only valid JSON arrays.' },
            { role: 'user', content: prompt },
          ],
          temperature: 0.3,
        },
      });

      const content = res.data.choices[0].message.content.trim();
      const jsonMatch = content.match(/\[[\s\S]*\]/);
      if (!jsonMatch) {
        throw new TranslationError('OpenAI response is not a valid JSON array', 'openai');
      }

      const translated = JSON.parse(jsonMatch[0]);
      if (!Array.isArray(translated) || translated.length !== texts.length) {
        throw new TranslationError(
          `OpenAI returned ${translated.length} translations for ${texts.length} texts`,
          'openai'
        );
      }

      return translated;
    } catch (error) {
      if (error instanceof TranslationError) throw error;
      const msg = error.response?.data?.error?.message || error.message;
      throw new TranslationError(`OpenAI API error: ${msg}`, 'openai', error);
    }
  }
}

async function translateBatch(texts, settings, from, to) {
  if (texts.length === 0) return []

  const { translationService, servicePriority, services } = settings.translationSetting
  const errors = []

  // Try primary service first
  const tryService = async (serviceName) => {
    const serviceConfig = services[serviceName]
    if (!serviceConfig) return null
    // Check if required config is present
    if (serviceName === 'baidu' && (!serviceConfig.appId || !serviceConfig.secretKey)) return null
    if (serviceName === 'tencent' && (!serviceConfig.secretId || !serviceConfig.secretKey)) return null
    if (serviceName === 'openai' && !serviceConfig.apiKey) return null
    if (serviceName === 'google' && !serviceConfig.apiKey) return null

    const provider = providerManager.resolveProvider(serviceName, serviceConfig)
    return await provider.translate(texts, shortLang(from, serviceName), shortLang(to, serviceName))
  }

  // Try primary
  try {
    const result = await tryService(translationService)
    if (result) return result
  } catch (error) {
    errors.push({ provider: translationService, error })
    console.log(chalk.yellow(`  ⚠ ${translationService} failed: ${error.message}`))
  }

  // Fallback by priority
  for (const serviceName of servicePriority) {
    if (serviceName === translationService) continue
    try {
      const result = await tryService(serviceName)
      if (result) return result
    } catch (error) {
      errors.push({ provider: serviceName, error })
      console.log(chalk.yellow(`  ⚠ ${serviceName} failed: ${error.message}`))
    }
  }

  throw new (require('./providers/errors').AllProvidersFailedError)(errors)
}
