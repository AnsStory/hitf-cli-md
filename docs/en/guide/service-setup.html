Configuration Example ​

json
{
  "translationSetting": {
    "translationService": "tencent",
    "services": {
      "tencent": {
        "secretId": "YOUR_SECRET_ID",
        "secretKey": "YOUR_SECRET_KEY",
        "region": "ap-guangzhou"
      },
      "baidu": {
        "appId": "YOUR_APP_ID",
        "secretKey": "YOUR_SECRET_KEY"
      },
      "openai": {
        "apiKey": "YOUR_API_KEY",
        "baseUrl": "https://api.openai.com",
        "model": "gpt-4o-mini"
      },
      "google": {
        "apiKey": "YOUR_API_KEY"
      }
    }
  }
}
